CommandBar.FocusLost:Connect(function(enterPressed)
	if not enterPressed then return end
	local input = CommandBar.Text
	CommandBar.Text = ""
	
	if input:sub(1, 1) ~= ";" then return end
	
	local arguments = string.split(input:sub(2), " ")
	local command = string.lower(arguments)
	local targetName = arguments
	
	AdminEvent:FireServer(command, targetName)
end)
