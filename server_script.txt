local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

local AdminEvent = ReplicatedStorage:FindFirstChild("AdminEvent")
if not AdminEvent then
	AdminEvent = Instance.new("RemoteEvent")
	AdminEvent.Name = "AdminEvent"
	AdminEvent.Parent = ReplicatedStorage
end

local AllowedAdmins = {
    [Your UserId here] = true,
    [UserId of a Moderator] = true,
}

local function findPlayer(nameString)
	if not nameString or nameString == "me" then return nil end
	for _, player in ipairs(Players:GetPlayers()) do
		if string.lower(player.Name):sub(1, #nameString) == string.lower(nameString) then
			return player
		end
	end
	return nil
end

AdminEvent.OnServerEvent:Connect(function(player, command, targetName, customReason)
	if not AllowedAdmins[player.UserId] then
		player:Kick("Unauthorized access to admin panel.")
		return
	end
	local targetPlayer = findPlayer(targetName)
	
	if command == "kill" and targetPlayer and targetPlayer.Character then
		local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
		if humanoid then humanoid.Health = 0 end
		
	elseif command == "kick" and targetPlayer then
		local kickMessage = customReason or "You have been kicked by an administrator."
		targetPlayer:Kick(kickMessage)
		
	elseif command == "punish" and targetPlayer and targetPlayer.Character then
		for _, part in ipairs(targetPlayer.Character:GetChildren()) do
			if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
				part.Transparency = 1
				part.CanCollide = false
			end
		end
	elseif command == "bring" and targetPlayer and targetPlayer.Character and player.Character then
		local targetRoot = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
		local localRoot = player.Character:FindFirstChild("HumanoidRootPart")
		if targetRoot and localRoot then
			targetRoot.CFrame = localRoot.CFrame * CFrame.new(0, 0, -3)
		end
		
	elseif command == "jail" and targetPlayer then
		if targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
			targetPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(0, -100, 0)
		end
		
	elseif command == "shutdown" then
		for _, p in ipairs(Players:GetPlayers()) do
			p:Kick("Server has been shut down by an administrator.")
		end
	end
end)
